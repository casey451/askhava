```sql
-- privacy (T862): her conversation address, the email she typed in her own text or web-chat conversation, sealed like a vault field and kept apart from her contact's email (PICK-029, INT-012, PRIV-001, PRIV-002, PRIV-007; D-5848, D-5972, D-5995).
-- privacy.conversation_email holds at most one address per contact: one she typed in a message
-- of her own conversation ("my email is ..."), taken where her raw words are scrubbed with her
-- contact in hand (privacy::api::read_inbound_for). An address from the quote page
-- (privacy.update_email) or shared at the page pick (privacy.contact.email) is never written
-- here. It is the only email the pick code may go to (PICK-029, D-050, D-5995) and is read only
-- by picks under the (Picks, PickCode) grant and by her own-record read under (PortalsHomeowner,
-- OwnRecord). The address is sealed with AES-256-GCM under VAULT_KEY, its associated data naming
-- the row ID, so a ciphertext copied to another row fails to open. A new address replaces the
-- row with a new ID. The contact purge deletes the row (it nulls the contact's fields, so a
-- cascade never fires then); the cascade covers a contact row deleted outright. The table is new
-- and empty, so nothing blocks.

CREATE TABLE privacy.conversation_email (
  id uuid PRIMARY KEY,
  contact_id uuid NOT NULL UNIQUE REFERENCES privacy.contact (id) ON DELETE CASCADE,
  email bytea NOT NULL,
  stored_at timestamptz NOT NULL
);

GRANT SELECT, INSERT, UPDATE, DELETE ON privacy.conversation_email TO askhava_app;
```

```rust
//! Her conversation address: the email she typed in her own text or web-chat conversation
//! (PICK-029, INT-012, TXT-034, PRIV-001, PRIV-002, PRIV-003, PRIV-007, SEC-046; T862, D-5848,
//! D-5972, D-5995).
//!
//! No web-chat step asks her email, and `privacy.contact.email` holds only the address she shared
//! at the page pick, which never gets a code (D-050). So an address she types in a message of
//! her own ("my email is ...", INT-012) is taken where her raw words are scrubbed with her contact
//! in hand (`read_inbound_for` calls [`capture`]); the scrubbed text is unchanged. D-5995: the
//! address she types at the web chat's update choice (INT-050, T878) is kept through
//! [`keep_conversation_address`] (D-5997) and may receive her pick code. It lives in `privacy.conversation_email`
//! (migration 0218), one row per contact, sealed under `VAULT_KEY` with associated data naming
//! the row ID (the `stated_name` sealing, as `privacy.update_email`), so a ciphertext moved to
//! another row fails to open. A quote-page or page-pick address is never written here.
//!
//! Two reads open it, each audited (the `email` field's name only, never the value), counted
//! for SEC-047 and committed before the plaintext leaves, on a bare connection (D-753):
//! [`read_conversation_for_pick_code`] for `picks`' pick code by email under the `(Picks,
//! PickCode)` grant, whose field list stays `[Phone]` so `privacy.contact.email` stays unreadable
//! under it; and [`read_own_conversation_email`] for her own record under `(PortalsHomeowner,
//! OwnRecord)`, which takes no contact ID but the proof's (SEC-046) and which T870 puts in her
//! download (PRIV-031). The contact purge deletes the row ([`forget_conversation_email`]).

/// The audit action of a stored conversation address (PICK-029, INT-012).
pub const CONVERSATION_EMAIL_STORED: &str = "conversation_email_stored";
/// The audit subject kind of a conversation address row.
const SUBJECT: &str = "privacy.conversation_email";

/// Her conversation address, opened. No `Debug`, `Display` or `Clone`: it is a contact value
/// and is never logged (PRIV-016).
pub struct ConversationAddress(SecretString);

impl ConversationAddress {
    /// The address, for `picks`' `ConversationEmail` (PICK-029) or her own download (T870).
    #[must_use]
    pub fn into_secret(self) -> SecretString {
        self.0
    }
}

/// INT-012's cues (D-5996), as folded words: an address is taken only as the next token after one.
const CUES: [&[&str]; 4] = [
    &["my", "email", "is"],
    &["my", "email", "address", "is"],
    &["mi", "correo", "es"],
    &["mi", "correo", "electronico", "es"],
];

/// INT-012, PICK-029, D-5848, D-5996: her own address in `raw`, taken only when one well-formed
/// address is the next token after an INT-012 cue (English or Spanish, case- and
/// accent-insensitive, matched on whole words so "amy email is" never counts, an optional `:`
/// after the cue). `None` when no cue precedes it, or when `raw` names any second, different
/// address anywhere, so a forwarded or third-party address never becomes the address that gets
/// her pick code. Every address in `raw` is counted from its scrubber key (`scrub::at_keys`), which
/// must pass the update address's check (`shaped`).
fn typed_address(raw: &str) -> Option<SecretString> {
    let mut all: Vec<SecretString> = Vec::new();
    for key in at_keys(raw) {
        let Some(address) = shaped(key.expose_secret()) else {
            continue;
        };
        if !all.iter().any(|f| f.expose_secret() == address.expose_secret()) {
            all.push(address);
        }
    }
    let [only] = all.as_slice() else {
        return None;
    };
    let words: Vec<String> = raw.split_whitespace().map(fold).collect();
    let next = cued_next(&words)?;
    let cued = shaped(next.trim_end_matches(['.', ',', ';', '!', '?']))?;
    (cued.expose_secret() == only.expose_secret()).then_some(cued)
}

/// The token right after the first whole-word cue in `words` (each already folded), skipping a
/// lone `:`; a cue's last word may carry the `:` itself ("is:").
fn cued_next(words: &[String]) -> Option<&str> {
    for start in 0..words.len() {
        for cue in CUES {
            let end = start + cue.len();
            let Some(window) = words.get(start..end) else {
                continue;
            };
            let last = cue.len() - 1;
            let hit = window.iter().zip(cue.iter()).enumerate().all(|(i, (w, c))| {
                w == c || (i == last && w.strip_suffix(':') == Some(*c))
            });
            if hit {
                let mut rest = words[end..].iter().map(String::as_str);
                let first = rest.next()?;
                return if first == ":" { rest.next() } else { Some(first) };
            }
        }
    }
    None
}

/// `word` lower-cased with the Spanish accents folded (á é í ó ú ü to the plain vowel), so a cue
/// matches whatever her keyboard typed.
fn fold(word: &str) -> String {
    word.to_lowercase()
        .chars()
        .map(|c| match c {
            'á' => 'a',
            'é' => 'e',
            'í' => 'i',
            'ó' => 'o',
            'ú' | 'ü' => 'u',
            other => other,
        })
        .collect()
}

/// `key` as an address when it has `UpdateAddress::parse`'s shape (TXT-018, D-4727).
fn shaped(key: &str) -> Option<SecretString> {
    let bare = bare_address(key).ok()?;
    let (name, domain) = bare.split_once('@')?;
    let ok = !name.is_empty()
        && !domain.contains('@')
        && domain.contains('.')
        && domain.split('.').all(|part| !part.is_empty());
    ok.then(|| SecretString::from(bare))
}

/// INT-012, PICK-029, PRIV-001, PRIV-002, D-5848, D-5995: stores the address in her message `raw`
/// as contact `contact`'s conversation address. It replaces an earlier, different one with a new
/// row ID and writes one audit row naming her and the `email` field, never the value. The same
/// address again, or no clear address, changes nothing. Runs as a savepoint inside
/// `read_inbound_for`'s connection, so it commits or rolls back with the caller.
///
/// # Errors
/// [`VaultError::Sealed`], [`VaultError::Audit`], [`VaultError::Database`].
pub(crate) async fn capture(
    db: &mut PgConnection,
    vault: &Vault,
    contact: Uuid,
    raw: &str,
) -> Result<(), VaultError> {
    let Some(address) = typed_address(raw) else {
        return Ok(());
    };
    let mut tx = db.begin().await.map_err(db_err)?;
    let held = sqlx::query!(
        "SELECT id, email FROM privacy.conversation_email WHERE contact_id = $1",
        contact
    )
    .fetch_optional(&mut *tx)
    .await
    .map_err(db_err)?;
    let same = held.is_some_and(|h| {
        vault
            .open_stated(h.id, &h.email)
            .is_ok_and(|old| old.expose_secret() == address.expose_secret())
    });
    if same {
        return Ok(());
    }
    let row = sqlx::query_scalar!(r#"SELECT gen_random_uuid() AS "id!""#)
        .fetch_one(&mut *tx)
        .await
        .map_err(db_err)?;
    let sealed = vault.seal_stated(row, address.expose_secret())?;
    sqlx::query!(
        "INSERT INTO privacy.conversation_email (id, contact_id, email, stored_at) \
         VALUES ($1, $2, $3, now()) ON CONFLICT (contact_id) DO UPDATE SET id = EXCLUDED.id, \
         email = EXCLUDED.email, stored_at = EXCLUDED.stored_at",
        row,
        contact,
        sealed,
    )
    .execute(&mut *tx)
    .await
    .map_err(db_err)?;
    record(&mut tx, (Actor::Homeowner(contact), CONVERSATION_EMAIL_STORED), row).await?;
    tx.commit().await.map_err(db_err)
}

/// The shared read: the grant (always registered: both pairs are in `GRANTS`), a bare
/// connection, then open, audit, count and commit before the address leaves (PRIV-003, D-753).
async fn open(
    db: &mut PgConnection,
    vault: &Vault,
    ((reader, purpose, fields), actor): ((Reader, Purpose, &[Field]), Actor),
    contact: Uuid,
) -> Result<Option<ConversationAddress>, VaultError> {
    let grant = grant_for(reader, purpose, fields).unwrap_or_else(|| unreachable!());
    ensure_bare(db).await?;
    let mut tx = db.begin().await.map_err(db_err)?;
    let Some(held) = sqlx::query!(
        "SELECT id, email FROM privacy.conversation_email WHERE contact_id = $1",
        contact
    )
    .fetch_optional(&mut *tx)
    .await
    .map_err(db_err)?
    else {
        return Ok(None);
    };
    let address = vault.open_stated(held.id, &held.email)?;
    record(&mut tx, (actor, grant.action), held.id).await?;
    count_read(&mut tx, grant.action).await?;
    tx.commit().await.map_err(db_err)?;
    Ok(Some(ConversationAddress(address)))
}

/// PRIV-007, D-5848: deletes contact `contact`'s conversation address, in the caller's
/// transaction (the contact purge's); how many rows went.
///
/// # Errors
/// [`VaultError::Database`].
pub(crate) async fn forget_conversation_email(
    db: &mut PgConnection,
    contact: Uuid,
) -> Result<u64, VaultError> {
    sqlx::query!(
        "DELETE FROM privacy.conversation_email WHERE contact_id = $1",
        contact
    )
    .execute(db)
    .await
    .map(|done| done.rows_affected())
    .map_err(db_err)
}

/// One audit row by `actor` for `action` on row `row`'s `email` field (PRIV-003, PRIV-028: the
/// field's name only).
async fn record(
    db: &mut PgConnection,
    (actor, action): (Actor, &'static str),
    row: Uuid,
) -> Result<(), VaultError> {
    let entry = AuditEntry {
        actor,
        action,
        subject_kind: SUBJECT,
        subject_id: row,
        change: Change::Field("email"),
    };
    audit(db, &entry)
        .await
        .map(drop)
        .map_err(|_| VaultError::Audit)
}
```
